include(CheckCXXCompilerFlag)

# Check the intended diagnostic, with a positive control for toolchain failures.
try_compile(HAS_NOTHROW_DESTRUCTOR_GUARD
    SOURCES ${CMAKE_CURRENT_SOURCE_DIR}/state_saver_destructor_compile_test.cpp
    CXX_STANDARD 11
    OUTPUT_VARIABLE DESTRUCTOR_CHECK_OUTPUT)
if(NOT HAS_NOTHROW_DESTRUCTOR_GUARD)
    message(FATAL_ERROR "state_saver positive compile check failed: ${DESTRUCTOR_CHECK_OUTPUT}")
endif()

try_compile(HAS_THROWING_DESTRUCTOR_GUARD
    SOURCES ${CMAKE_CURRENT_SOURCE_DIR}/state_saver_destructor_compile_test.cpp
    CXX_STANDARD 11
    COMPILE_DEFINITIONS -DTEST_THROWING_DESTRUCTOR
    OUTPUT_VARIABLE DESTRUCTOR_CHECK_OUTPUT)
if(HAS_THROWING_DESTRUCTOR_GUARD OR NOT DESTRUCTOR_CHECK_OUTPUT MATCHES "state_saver requires nothrow destructible type")
    message(FATAL_ERROR "state_saver must reject throwing destructors: ${DESTRUCTOR_CHECK_OUTPUT}")
endif()

function(configure_test target std)
    target_compile_options(${target} PRIVATE ${OPTIONS})
    target_compile_definitions(${target} PRIVATE DOCTEST_CONFIG_USE_STD_HEADERS)
    target_include_directories(${target} PRIVATE 3rdparty/doctest)
    target_link_libraries(${target} PRIVATE ${PROJECT_NAME}::${PROJECT_NAME})
    set_target_properties(${target} PROPERTIES CXX_EXTENSIONS OFF)
    if(std)
        if(MSVC)
            target_compile_options(${target} PRIVATE /std:${std})
        else()
            target_compile_options(${target} PRIVATE -std=${std})
        endif()
    endif()
    add_test(NAME ${target} COMMAND ${target})
endfunction()

function(make_test target std)
    add_executable(${target} ${SOURCES})
    configure_test(${target} "${std}")
endfunction()

function(make_mode_test target source std)
    add_executable(${target} ${source})
    configure_test(${target} "${std}")
endfunction()

# Each public header must compile without another header supplying its dependencies.
foreach(header attributes byte concepts state_saver type_traits unused utility)
    set(header_std 11)
    if(header STREQUAL "attributes")
        set(header_std 98)
    elseif(header STREQUAL "byte" OR header STREQUAL "utility")
        set(header_std 17)
    endif()
    set(source "${CMAKE_CURRENT_BINARY_DIR}/${header}_standalone.cpp")
    file(WRITE "${source}" "#include <${header}.hpp>\n")
    set(target "${PROJECT_NAME}-${header}-standalone")
    add_library(${target} OBJECT "${source}")
    target_link_libraries(${target} PRIVATE ${PROJECT_NAME}::${PROJECT_NAME})
    target_compile_options(${target} PRIVATE ${OPTIONS})
    set_target_properties(${target} PROPERTIES CXX_STANDARD ${header_std} CXX_STANDARD_REQUIRED ON CXX_EXTENSIONS OFF)
endforeach()
